    pub(crate) fn probe_keyboard_enhancement(io: &mut dyn ProbeIo, bound: Duration) -> std::io::Result<bool> {
        io.write(b"\x1b[?u\x1b[c")?;  // Request kitty (u variant) and DA1
        let deadline = Instant::now() + bound;
        let mut buffer = Vec::new();
        
        loop {
            let remaining = deadline.saturating_duration_since(Instant::now());
            if remaining.is_zero() { return Ok(false); }
            
            match io.read_with_timeout(remaining)? {
                Some(chunk) => {
                    buffer.extend_from_slice(&chunk);
                    // Parse: look for ESC [ ? <digits> u (u variant = true)
                    if buffer.contains(&b'u') && buffer.windows(3).any(|w| w == b"[?") {
                        if buffer.iter().skip_while(|&&b| b != b'?').any(|&b| b.is_ascii_digit()) {
                            return Ok(true);
                        }
                    }
                    // Parse: look for ESC [ ... c without u (DA1 only = false)
                    if buffer.contains(&b'c') && !buffer.contains(&b'u') {
                        return Ok(false);
                    }
                }
                None => return Ok(false),
            }
        }
    }

| Durable writes during restore | Write batching applies only during restore Phase A; interactive saves outside restore write immediately. Failure semantics unchanged. |
